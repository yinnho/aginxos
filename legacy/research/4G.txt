3. 核心代码：4G 拨号（QMI）

src/main.rs
use std::fs::OpenOptions;
use std::os::unix::fs::OpenOptionsExt;
use std::io::{Read, Write};

use nix::ioctl;

const QMI_IOCTL: u32 = 0x8be0050b; // QMI 标准 ioctl

// 极简 QMI 原始报文（激活 PDP 上下文，移动/联通/电信通用）
const QMI_RAW_PKT: &[u8] = &[
    0x01, 0x00, 0x00, 0x00, 0x01, 0x00, 0x00, 0x00,
    0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
    0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
    0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
];

fn main() -> Result<(), Box<dyn std::error::Error>> {
    println!("Pixel 5 4G 拨号启动...");

    // 1. 打开 QMI 设备
    let mut qmi_dev = OpenOptions::new()
        .read(true)
        .write(true)
        .custom_flags(libc::O_SYNC)
        .open("/dev/cdc-wdm0")?;

    // 2. 发送 QMI 激活
    qmi_dev.write_all(QMI_RAW_PKT)?;
    let mut buf = [0u8; 512];
    let n = qmi_dev.read(&mut buf)?;
    println!("QMI 响应长度: {}", n);

    // 3. 启用 rmnet0
    run_shell("ip link set dev rmnet0 up")?;
    run_shell("dhclient rmnet0")?;
    // 或者静态IP（如果运营商不给DHCP）
    // run_shell("ip addr add 10.0.0.2/24 dev rmnet0")?;
    // run_shell("ip route add default via 10.0.0.1")?;

    println!("✅ 4G 联网完成");
    Ok(())
}

// 极简 shell 调用（Root 可用）
fn run_shell(cmd: &str) -> Result<(), Box<dyn std::error::Error>> {
    let _ = std::process::Command::new("sh")
        .arg("-c")
        .arg(cmd)
        .status()?;
    Ok(())
}

// 示例：PMIC 给基带电源上电（裸机真实操作）
const MPM_SPMI_BASE: usize = 0x0C10_0000;
const MPM_PWR_EN: *mut u32 = (MPM_SPMI_BASE + 0x100) as _;

unsafe fn modem_power_on() {
    *MPM_PWR_EN |= 1 << 5; // 基带电源使能
    cortex_a::asm::nop();
    cortex_a::delay::msleep(300);
}
3. UART 发 AT 指令（真实可用）
// 裸机 UART 输出
unsafe fn uart_puts(s: &str) {
    for &c in s.as_bytes() {
        const UART_BASE: usize = 0x00A9C000; // Pixel5 UART2
        *(UART_BASE as *mut u32) = c as u32;
    }
}

// AT 拨号上网（裸机最稳的 4G 方式）
unsafe fn modem_dial() {
    uart_puts("AT\r\n");
    cortex_a::delay::msleep(100);

    uart_puts("AT+CGDCONT=1,\"IP\",\"cmnet\"\r\n");
    cortex_a::delay::msleep(200);

    uart_puts("ATD*99#\r\n"); // 拨号
    cortex_a::delay::msleep(1000);
}
拨号成功后，基带会进入 PPP 模式，开始发 IP 包。
